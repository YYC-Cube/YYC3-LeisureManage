# 03 · 数据库设计(MySQL 8)

> 版本:v1.1(2026-11-01 裁决修订)|**定位裁决:独立运营/多门店,完全脱离「智慧商家」**——external_id 语义改为"开放生态预留",artist_order 由"主系统镜像"改为自有订单,联动同步注释废止
> 版本:v1.0(2026-11-01)|字段依据:App 25 子模块实测表单/列表列名 + PC 配置项(历史考察取证)
> 约定:InnoDB / utf8mb4;全部业务表含 `store_id`(多门店数据隔离)、`business_date`(营业日冗余)、审计四件套 `created_by/created_at/updated_by/updated_at`;软删用 `deleted_at`。

## 1. ER 总览

```
store 1─n artist_group 1─n artist n─1 artist_level
artist 1─1 artist_wechat_bind
artist 1─n attendance_report        (报备=签到/签退两条或单条含签退)
artist 1─n show_session(下钟计费计入房账,方案甲) ; room_session 1─n stage_fee_txn(ROOM_SETTLE 关房结算)
artist_level n─1 stage_fee_rule n─1 room_type
artist 1─n stage_fee_txn(充值/手动扣费)
artist 1─n ticket_txn(购买/使用)   n─1 ticket_rule
artist 1─n artist_review / artist_booking
artist 1─n artist_order(自有订单,独立运营闭环)
admin_user n─n role n─n permission;audit_log 独立追加写
```

## 2. DDL

```sql
-- 门店(本地维护;多门店数据隔离的锚点表)
CREATE TABLE store (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(64) NOT NULL COMMENT '门店名(历史考察样例:皇家国际)',
  address VARCHAR(255), city VARCHAR(64),
  business_day_cut_time TIME NOT NULL DEFAULT '08:00:00' COMMENT '营业日切点(门店级可配)',
  external_id VARCHAR(64) COMMENT '外部生态ID(开放API预留;独立运营可不填)',
  status TINYINT DEFAULT 1,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  UNIQUE KEY uk_external (external_id)
) COMMENT '门店';

-- 包厢类型(台费规则维度;门店自维护)
CREATE TABLE room_type (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  name VARCHAR(32) NOT NULL COMMENT '实测:小包/中包/大包/VIP包/西亳厅/外卖房',
  external_id VARCHAR(64),
  sort INT DEFAULT 0,
  status TINYINT DEFAULT 1,
  UNIQUE KEY uk_store_name (store_id, name)
) COMMENT '包厢类型';

-- 销售经理组别
CREATE TABLE artist_group (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  name VARCHAR(64) NOT NULL COMMENT '实测以主管命名:江海龙',
  supervisor_name VARCHAR(64) COMMENT '主管',
  deleted_at DATETIME NULL,
  UNIQUE KEY uk_store_name (store_id, name, deleted_at)
) COMMENT '销售经理组别';

-- 销售经理等级
CREATE TABLE artist_level (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  name VARCHAR(64) NOT NULL COMMENT '实测:技师',
  deleted_at DATETIME NULL,
  UNIQUE KEY uk_store_name (store_id, name, deleted_at)
) COMMENT '销售经理等级';

-- 销售经理档案(核心)
CREATE TABLE artist (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  group_id BIGINT UNSIGNED COMMENT '组别*',
  supervisor_name VARCHAR(64),
  level_id BIGINT UNSIGNED COMMENT '等级*',
  real_name VARCHAR(64) NOT NULL COMMENT '姓名*',
  stage_name VARCHAR(64) COMMENT '艺名(检索主键)',
  gender TINYINT DEFAULT 1 COMMENT '性别* 1女2男(实测默认女)',
  phone VARCHAR(20),
  badge_no VARCHAR(32) COMMENT '腰牌号(店内唯一)',
  id_card_no_enc VARBINARY(512) COMMENT '身份证号(AES加密)',
  id_card_img_key VARCHAR(255) COMMENT '身份证照OSS键(加密)',
  avatar_keys JSON COMMENT '人像照片OSS键数组',
  tags JSON COMMENT '标签数组',
  performance_target DECIMAL(12,2) DEFAULT 0 COMMENT '业绩指标(元,周期考核)',
  max_total_shows INT DEFAULT 0 COMMENT '总上台次数上限,0=不限',
  max_cross_rooms INT DEFAULT 0 COMMENT '允许串台数,0=不允许',
  first_show_min_minutes INT DEFAULT 0 COMMENT '首台最低上钟时长(分)',
  same_room_min_minutes INT DEFAULT 0 COMMENT '上钟最低时长(同包厢,分)',
  work_start_mode TINYINT DEFAULT 0 COMMENT '0自由时间1固定',
  work_start TIME NULL, work_end TIME NULL,
  work_end_mode TINYINT DEFAULT 0,
  tip_fixed DECIMAL(10,2) DEFAULT 0 COMMENT '小费一口价',
  hourly_rate DECIMAL(10,2) DEFAULT 0 COMMENT '计时单价',
  wechat_openid VARCHAR(64) NULL COMMENT '微信唯一绑定',
  status TINYINT DEFAULT 1 COMMENT '1在职0停用2已转移',
  deleted_at DATETIME NULL,
  UNIQUE KEY uk_badge (store_id, badge_no),
  UNIQUE KEY uk_openid (wechat_openid),
  KEY idx_store_stage (store_id, stage_name),
  KEY idx_store_group (store_id, group_id)
) COMMENT '销售经理档案';

-- 报备(考勤)
CREATE TABLE attendance_report (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  business_date DATE NOT NULL,
  artist_id BIGINT UNSIGNED NOT NULL,
  group_id BIGINT UNSIGNED, supervisor_name VARCHAR(64),
  type VARCHAR(16) NOT NULL COMMENT 'CHECKIN/陪宴/私事/早退/迟到/例假/其他/未报备',
  report_time DATETIME NOT NULL COMMENT '报备时间',
  checkin_lat DECIMAL(10,6), checkin_lng DECIMAL(10,6),
  checkin_geo_ok TINYINT DEFAULT 1 COMMENT '定位是否可用',
  checkout_time DATETIME NULL,
  checkout_lat DECIMAL(10,6), checkout_lng DECIMAL(10,6),
  source TINYINT DEFAULT 0 COMMENT '0自助扫码1补打卡',
  operator_id BIGINT UNSIGNED COMMENT '补卡人',
  remark VARCHAR(255),
  KEY idx_store_date (store_id, business_date),
  KEY idx_artist_date (artist_id, business_date)
) COMMENT '报备考勤';

-- 上下钟场次
CREATE TABLE show_session (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  business_date DATE NOT NULL,
  artist_id BIGINT UNSIGNED NOT NULL,
  room_id BIGINT UNSIGNED, room_type_id BIGINT UNSIGNED,
  room_name VARCHAR(64) COMMENT '包厢名(实测8311/四楼外卖房)',
  start_time DATETIME NOT NULL COMMENT '上钟',
  end_time DATETIME NULL COMMENT '下钟',
  duration_minutes INT NULL,
  is_cross TINYINT DEFAULT 0 COMMENT '串台标记',
  fee_amount DECIMAL(10,2) NULL COMMENT '计得台费',
  bill_no VARCHAR(64) COMMENT '关联账单号(联动)',
  operator_id BIGINT UNSIGNED,
  KEY idx_store_date (store_id, business_date),
  KEY idx_artist (artist_id, start_time)
) COMMENT '上下钟场次';

-- 台费规则
CREATE TABLE stage_fee_rule (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  level_id BIGINT UNSIGNED NOT NULL COMMENT '销售经理等级',
  room_type_ids JSON NOT NULL COMMENT '适用包厢类型数组',
  amount DECIMAL(10,2) NOT NULL DEFAULT 0 COMMENT '台费金额',
  deleted_at DATETIME NULL,
  KEY idx_store_level (store_id, level_id)
) COMMENT '台费规则(等级×房型)';

-- 台费流水(不可变)
CREATE TABLE stage_fee_txn (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  business_date DATE NOT NULL,
  artist_id BIGINT UNSIGNED NOT NULL,
  type VARCHAR(16) NOT NULL COMMENT 'RECHARGE充值/ROOM_SETTLE房账关房结算(方案甲)/MANUAL_DEDUCT手动扣费/REVERSE冲正',
  amount DECIMAL(10,2) NOT NULL COMMENT '充值+,扣费-',
  pay_channel VARCHAR(16) COMMENT 'CASH现金/DOUYIN抖音(充值时)',
  pay_txn_no VARCHAR(64) COMMENT '三方支付单号',
  ref_type VARCHAR(16) COMMENT '来源:session/manual/reverse',
  ref_id BIGINT UNSIGNED,
  remark VARCHAR(255) COMMENT '手动扣费必填',
  approval_id BIGINT UNSIGNED COMMENT '高额审批单',
  operator_id BIGINT UNSIGNED,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  KEY idx_store_date (store_id, business_date),
  KEY idx_artist_date (artist_id, business_date)
) COMMENT '台费流水(只增不改)' -- REVOKED UPDATE/DELETE

-- 台票规则
CREATE TABLE ticket_rule (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  name VARCHAR(64) NOT NULL COMMENT '台票名称',
  price DECIMAL(10,2) NOT NULL,
  valid_days INT DEFAULT 0 COMMENT '有效期(0=长期)',
  status TINYINT DEFAULT 1, deleted_at DATETIME NULL
) COMMENT '台票规则';

-- 台票流水(不可变)
CREATE TABLE ticket_txn (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  business_date DATE NOT NULL,
  artist_id BIGINT UNSIGNED NOT NULL,
  type VARCHAR(16) NOT NULL COMMENT 'BUY购买/USE使用/EXPIRE过期',
  ticket_rule_id BIGINT UNSIGNED,
  qty INT DEFAULT 1,
  amount DECIMAL(10,2) DEFAULT 0 COMMENT '购买金额',
  pay_channel VARCHAR(16),
  ref_session_id BIGINT UNSIGNED COMMENT '使用时关联场次',
  operator_id BIGINT UNSIGNED,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  KEY idx_store_date (store_id, business_date),
  KEY idx_artist (artist_id, business_date)
) COMMENT '台票流水(只增不改)';

-- 销售经理订单(自有订单,独立运营闭环)
CREATE TABLE artist_order (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  business_date DATE NOT NULL,
  artist_id BIGINT UNSIGNED,
  order_no VARCHAR(64) NOT NULL COMMENT '实测:KB01665…',
  order_type VARCHAR(32) COMMENT '酒水订单/计时开房/台费/台票…',
  pay_status VARCHAR(16) COMMENT '待支付/已付款',
  pay_type VARCHAR(16) COMMENT '后买单/现金/抖音…',
  amount DECIMAL(10,2),
  paid_at DATETIME NULL,
  external_id VARCHAR(64),
  UNIQUE KEY uk_external (external_id),
  KEY idx_store_date (store_id, business_date)
) COMMENT '销售经理订单(联动镜像)';

-- 服务评价
CREATE TABLE artist_review (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  business_date DATE NOT NULL,
  artist_id BIGINT UNSIGNED NOT NULL,
  review_type VARCHAR(32) COMMENT '评价类型(实测:全部评价下拉)',
  score TINYINT, content VARCHAR(500),
  reviewer VARCHAR(64) COMMENT '顾客昵称/手机尾号',
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  KEY idx_artist_date (artist_id, business_date)
) COMMENT '销售经理服务评价';

-- 预约单
CREATE TABLE artist_booking (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  business_date DATE NOT NULL,
  booking_no VARCHAR(32) NOT NULL COMMENT '预约号',
  booking_time DATETIME COMMENT '时间',
  room_id BIGINT UNSIGNED, room_name VARCHAR(64) COMMENT '包厢',
  group_id BIGINT UNSIGNED, artist_id BIGINT UNSIGNED,
  artist_name VARCHAR(64) COMMENT '艺名/腰牌号',
  status VARCHAR(16) DEFAULT 'BOOKED' COMMENT 'BOOKED/ARRIVED/CANCELLED/NO_SHOW',
  UNIQUE KEY uk_no (store_id, booking_no),
  KEY idx_store_date (store_id, business_date)
) COMMENT '销售经理预约';

-- 展示投放(展示控制台)
CREATE TABLE showcase_cast (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  room_id BIGINT UNSIGNED COMMENT '当前包厢(NULL=全局)',
  group_id BIGINT UNSIGNED COMMENT '当前组别',
  artist_id BIGINT UNSIGNED NOT NULL COMMENT '被投放销售经理(被选中态)',
  cast_by BIGINT UNSIGNED, created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  ended_at DATETIME NULL
) COMMENT '展示投放(被选中态来源)';

-- 审批单(手动扣费高额)
CREATE TABLE approval (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  biz_type VARCHAR(16) NOT NULL COMMENT 'MANUAL_DEDUCT…',
  biz_payload JSON NOT NULL,
  amount DECIMAL(10,2),
  status VARCHAR(16) DEFAULT 'PENDING' COMMENT 'PENDING/APPROVED/REJECTED',
  applicant_id BIGINT UNSIGNED, approver_id BIGINT UNSIGNED,
  decided_at DATETIME NULL, remark VARCHAR(255),
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
) COMMENT '审批单';

-- 管理员/RBAC
CREATE TABLE admin_user (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED, name VARCHAR(64) NOT NULL,
  phone VARCHAR(20) UNIQUE, password_hash VARCHAR(128),
  role_id BIGINT UNSIGNED, status TINYINT DEFAULT 1,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
) COMMENT '管理端用户';
CREATE TABLE role (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED, name VARCHAR(64),
  perm_codes JSON COMMENT '功能码数组'
) COMMENT '角色(功能映射可配,对齐实测权限矩阵)';

-- 审计(追加写,无更新权限)
CREATE TABLE audit_log (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED, actor_id BIGINT UNSIGNED, actor_name VARCHAR(64),
  action VARCHAR(64) NOT NULL COMMENT 'ARTIST_CREATE/FEE_MANUAL/…',
  target_type VARCHAR(32), target_id VARCHAR(64),
  before_json JSON, after_json JSON,
  ip VARCHAR(45), ua VARCHAR(255),
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  KEY idx_store_time (store_id, created_at)
) COMMENT '审计日志(只增)';

-- 日切汇总(报表加速,日切任务物化)
CREATE TABLE daily_summary (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  store_id BIGINT UNSIGNED NOT NULL,
  business_date DATE NOT NULL,
  metrics JSON NOT NULL COMMENT '{充值,扣费,上台人次,串台次数,报备各状态人数,台票购买/使用…}',
  UNIQUE KEY uk_store_date (store_id, business_date)
) COMMENT '营业日汇总';
```

## 3. 关键约束与策略

| 策略 | 实现 |
|---|---|
| 流水不可变 | `stage_fee_txn/ticket_txn/audit_log` 账号 revoke UPDATE/DELETE;冲正走 REVERSE |
| 腰牌号唯一 | `uk_badge(store_id,badge_no)` |
| 微信唯一 | `uk_openid`;跨店转移=事务内:旧店 status=2已转移+audit,新店建档+绑定 |
| 营业日 | 写入时由服务层按 store.business_day_cut_time 计算,冗余存储 |
| 台费快照 | show_session.fee_amount 冗余成交时规则值(规则后续改动不影响历史) |
| 敏感字段 | id_card_no_enc AES-256-GCM;密钥 KMS;照片 OSS 私有+签名 URL 短时效 |
| 分区/归档 | txn 类表按月 RANGE 分区;3 年前归档表 |
| 汇总一致 | daily_summary 由日切任务重算+校验(明细和=汇总) |
